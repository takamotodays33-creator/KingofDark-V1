local U = {}

function U.getChar() return LP.Character or LP.CharacterAdded:Wait() end

function U.getHRP()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

function U.getHum()
    local c = LP.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end

function U.tw(obj, time, props, style, dir)
    if not obj or not obj.Parent then return end
    local t = TweenService:Create(
        obj,
        TweenInfo.new(time or 0.22, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

function U.safe(fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok then warn("[KingOfDark] " .. tostring(err)) end
    return ok
end

function U.clamp(v, mn, mx)
    return math.max(mn, math.min(mx, v))
end

function U.round(n)
    return math.floor(n + 0.5)
end

function U.log(tag, msg)
    local entry = string.format("[%s] [%s] %s", os.date("%H:%M:%S"), tag, msg)
    table.insert(KOD.logs, entry)
    if #KOD.logs > 200 then table.remove(KOD.logs, 1) end
    print("[KingOfDark] " .. entry)
end

local Save = {
    path = "KingOfDark/config.json",
    data = {},
}

function Save.load()
    if not isfile then return {} end
    local ok, exists = pcall(isfile, Save.path)
    if not ok or not exists then return {} end
    local ok2, content = pcall(readfile, Save.path)
    if not ok2 or not content then return {} end
    local ok3, decoded = pcall(HttpService.JSONDecode, HttpService, content)
    if not ok3 then return {} end
    return decoded or {}
end

function Save.write()
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, Save.data)
    if not ok then return end
    pcall(writefile, Save.path, encoded)
end

Save.data = Save.load()
U.log("Core", "Persistência carregada: " .. tostring(next(Save.data) ~= nil))

-- ═══════════════════════════════════════════════════════════════
local function makeDraggable(frame, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
makeDraggable(Main, TitleBar)
makeDraggable(FloatBtn, FloatBtn)

function TabSystem.select(key)
    if TabSystem.active == key then return end

    -- Desativa anterior
    if TabSystem.active and TabSystem.tabs[TabSystem.active] then
        local prev = TabSystem.tabs[TabSystem.active]
        prev.page.Visible = false
        U.tw(prev.button, 0.18, { BackgroundTransparency = 1, BackgroundColor3 = T.Glass })
        U.tw(prev.nameLbl, 0.18, { TextColor3 = T.TextDim })
        U.tw(prev.iconLbl, 0.18, { TextColor3 = T.TextDim })
        U.tw(prev.innerGlow, 0.18, { Size = UDim2.new(0, 3, 0, 0) })
    end

    -- Ativa novo
    local tab = TabSystem.tabs[key]
    if not tab then return end

    tab.page.Visible = true
    U.tw(tab.button, 0.18, { BackgroundTransparency = 0.15, BackgroundColor3 = T.Deep })
    U.tw(tab.nameLbl, 0.18, { TextColor3 = T.Text })
    U.tw(tab.iconLbl, 0.18, { TextColor3 = T.Glow })
    U.tw(tab.innerGlow, 0.22, { Size = UDim2.new(0, 3, 0, 24) })

    TabSystem.active = key

    -- Fade in da página
    tab.page.ScrollBarImageTransparency = 0
    for _, child in ipairs(tab.page:GetChildren()) do
        if child:IsA("GuiObject") and not child:IsA("UIListLayout") and not child:IsA("UIPadding") then
            child.BackgroundTransparency = 1
            local target = child:GetAttribute("targetBG") or 0.15
            U.tw(child, 0.25, { BackgroundTransparency = target })
        end
    end

    U.log("Tabs", "Aba ativa: " .. tab.name)
end

function Components.slider(parent, opts)
    opts = opts or {}
    local text    = opts.text or "Slider"
    local min     = opts.min or 0
    local max     = opts.max or 100
    local default = opts.default or min
    local suffix  = opts.suffix or ""
    local decimals = opts.decimals or 0
    local onChange = opts.callback
    local desc    = opts.desc

    local row = Instance.new("Frame")
    row.Name = "Slider"
    row.Size = UDim2.new(1, 0, 0, desc and 74 or 60)
    row.BackgroundColor3 = T.Card
    row.BackgroundTransparency = 0.15
    row.BorderSizePixel = 0
    row.Parent = parent
    Instance.new("UICorner", row).CornerRadius = UDim.new(0, 10)

    local stroke = Instance.new("UIStroke")
    stroke.Color = T.Border
    stroke.Thickness = 1
    stroke.Transparency = 0.7
    stroke.Parent = row

    local label = Instance.new("TextLabel")
    label.Text = text
    label.Font = FONT_MED
    label.TextSize = 12
    label.TextColor3 = T.Text
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Size = UDim2.new(1, -100, 0, 18)
    label.Position = UDim2.new(0, 14, 0, 6)
    label.BackgroundTransparency = 1
    label.Parent = row

    if desc then
        local d = Instance.new("TextLabel")
        d.Text = desc
        d.Font = FONT
        d.TextSize = 10
        d.TextColor3 = T.TextMuted
        d.TextXAlignment = Enum.TextXAlignment.Left
        d.Size = UDim2.new(1, -100, 0, 12)
        d.Position = UDim2.new(0, 14, 0, 22)
        d.BackgroundTransparency = 1
        d.Parent = row
    end

    local valueBox = Instance.new("TextBox")
    valueBox.Text = tostring(default) .. suffix
    valueBox.Font = FONT_BOLD
    valueBox.TextSize = 11
    valueBox.TextColor3 = T.Accent
    valueBox.BackgroundColor3 = T.GlassHi
    valueBox.BorderSizePixel = 0
    valueBox.Size = UDim2.new(0, 68, 0, 22)
    valueBox.Position = UDim2.new(1, -80, 0, 8)
    valueBox.ClearTextOnFocus = false
    valueBox.Parent = row
    Instance.new("UICorner", valueBox).CornerRadius = UDim.new(0, 6)

    local barY = desc and 50 or 38

    local barBg = Instance.new("Frame")
    barBg.Size = UDim2.new(1, -28, 0, 6)
    barBg.Position = UDim2.new(0, 14, 0, barY)
    barBg.BackgroundColor3 = T.GlassHi
    barBg.BorderSizePixel = 0
    barBg.Parent = row
    Instance.new("UICorner", barBg).CornerRadius = UDim.new(0, 3)

    local initPct = (default - min) / (max - min)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(initPct, 0, 1, 0)
    fill.BackgroundColor3 = T.Accent
    fill.BorderSizePixel = 0
    fill.Parent = barBg
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 3)

    local fillGlow = Instance.new("UIStroke")
    fillGlow.Color = T.Glow
    fillGlow.Thickness = 1
    fillGlow.Transparency = 0.5
    fillGlow.Parent = fill

    local knobDot = Instance.new("Frame")
    knobDot.Size = UDim2.new(0, 14, 0, 14)
    knobDot.Position = UDim2.new(initPct, -7, 0.5, -7)
    knobDot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knobDot.BorderSizePixel = 0
    knobDot.ZIndex = 3
    knobDot.Parent = barBg
    Instance.new("UICorner", knobDot).CornerRadius = UDim.new(1, 0)

    local knobStroke = Instance.new("UIStroke")
    knobStroke.Color = T.Accent
    knobStroke.Thickness = 2
    knobStroke.Parent = knobDot

    local dragging = false
    local currentValue = default

    local barBtn = Instance.new("TextButton")
    barBtn.Text = ""
    barBtn.Size = UDim2.new(1, 0, 3, 0)
    barBtn.Position = UDim2.new(0, 0, -1, 0)
    barBtn.BackgroundTransparency = 1
    barBtn.Parent = barBg

    local function formatVal(v)
        if decimals > 0 then
            return string.format("%." .. decimals .. "f", v) .. suffix
        end
        return tostring(math.floor(v)) .. suffix
    end

    local function updateFromInput(input)
        local relX = U.clamp((input.Position.X - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
        local raw = min + (max - min) * relX
        local val
        if decimals > 0 then
            local mult = 10 ^ decimals
            val = math.floor(raw * mult + 0.5) / mult
        else
            val = math.floor(raw + 0.5)
        end
        currentValue = val
        fill.Size = UDim2.new(relX, 0, 1, 0)
        knobDot.Position = UDim2.new(relX, -7, 0.5, -7)
        valueBox.Text = formatVal(val)
        if onChange then U.safe(onChange, val) end
    end

    barBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    valueBox.FocusLost:Connect(function()
        local n = tonumber(valueBox.Text:match("[-%d%.]+"))
        if n then
            n = U.clamp(n, min, max)
            local relX = (n - min) / (max - min)
            fill.Size = UDim2.new(relX, 0, 1, 0)
            knobDot.Position = UDim2.new(relX, -7, 0.5, -7)
            currentValue = n
            valueBox.Text = formatVal(n)
            if onChange then U.safe(onChange, n) end
        else
            valueBox.Text = formatVal(currentValue)
        end
    end)

    return {
        frame = row,
        get = function() return currentValue end,
        set = function(v)
            v = U.clamp(v, min, max)
            local relX = (v - min) / (max - min)
            fill.Size = UDim2.new(relX, 0, 1, 0)
            knobDot.Position = UDim2.new(relX, -7, 0.5, -7)
            valueBox.Text = formatVal(v)
            currentValue = v
        end,
    }
end

function Components.button(parent, opts)
    opts = opts or {}
    local text = opts.text or "Botão"
    local style = opts.style or "default"
    local callback = opts.callback
    local icon = opts.icon

    local btn = Instance.new("TextButton")
    btn.Text = icon and (icon .. "  " .. text) or text
    btn.Font = FONT_BOLD
    btn.TextSize = 12
    btn.TextColor3 = T.Text
    btn.BackgroundColor3 = T.Card
    btn.BackgroundTransparency = 0.15
    btn.BorderSizePixel = 0
    btn.AutoButtonColor = false
    btn.Size = UDim2.new(1, 0, 0, 40)
    btn.Parent = parent
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)

    local stroke = Instance.new("UIStroke")
    stroke.Color = T.Border
    stroke.Thickness = 1
    stroke.Transparency = 0.7
    stroke.Parent = btn

    if style == "primary" then
        btn.BackgroundColor3 = T.Deep
        btn.BackgroundTransparency = 0.05
        stroke.Color = T.Accent
        stroke.Transparency = 0.3
    elseif style == "danger" then
        btn.BackgroundColor3 = T.Danger
        btn.BackgroundTransparency = 0.7
        stroke.Color = T.Danger
        stroke.Transparency = 0.4
        btn.TextColor3 = T.Danger
    elseif style == "success" then
        btn.BackgroundColor3 = T.Success
        btn.BackgroundTransparency = 0.75
        stroke.Color = T.Success
        stroke.Transparency = 0.4
        btn.TextColor3 = T.Success
    end

    btn.MouseEnter:Connect(function()
        if style == "danger" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Danger, BackgroundTransparency = 0.5, TextColor3 = Color3.fromRGB(255,255,255) })
        elseif style == "success" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Success, BackgroundTransparency = 0.5, TextColor3 = Color3.fromRGB(255,255,255) })
        elseif style == "primary" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Accent, BackgroundTransparency = 0.2 })
        else
            U.tw(btn, 0.15, { BackgroundColor3 = T.Deep, BackgroundTransparency = 0.1 })
        end
        U.tw(stroke, 0.15, { Color = T.Accent, Transparency = 0.2 })
    end)
    btn.MouseLeave:Connect(function()
        if style == "danger" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Danger, BackgroundTransparency = 0.7, TextColor3 = T.Danger })
        elseif style == "success" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Success, BackgroundTransparency = 0.75, TextColor3 = T.Success })
        elseif style == "primary" then
            U.tw(btn, 0.15, { BackgroundColor3 = T.Deep, BackgroundTransparency = 0.05 })
        else
            U.tw(btn, 0.15, { BackgroundColor3 = T.Card, BackgroundTransparency = 0.15 })
        end
        U.tw(stroke, 0.15, { Color = T.Border, Transparency = 0.7 })
    end)

    btn.MouseButton1Click:Connect(function()
        U.tw(btn, 0.08, { BackgroundTransparency = 0.5 })
        task.wait(0.08)
        U.tw(btn, 0.15, { BackgroundTransparency = style == "primary" and 0.05 or 0.15 })
        if callback then U.safe(callback) end
    end)

    return {
        frame = btn,
        setText = function(t) btn.Text = icon and (icon .. "  " .. t) or t end,
    }
end

local Target = {}

function Target.getPlayers()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                table.insert(list, {
                    player = p,
                    char = p.Character,
                    hrp = hrp,
                    hum = hum,
                    distance = (hrp.Position - (U.getHRP() and U.getHRP().Position or Vector3.zero)).Magnitude,
                    health = hum.Health,
                    maxHealth = hum.MaxHealth,
                })
            end
        end
    end
    return list
end

function Target.nearest()
    local list = Target.getPlayers()
    table.sort(list, function(a, b) return a.distance < b.distance end)
    return list[1]
end

function Target.lowestHP()
    local list = Target.getPlayers()
    table.sort(list, function(a, b) return a.health < b.health end)
    return list[1]
end

function Target.random()
    local list = Target.getPlayers()
    return list[math.random(1, math.max(1, #list))]
end

-- ═══════════════════════════════════════════════════════════════
--  [36] SISTEMA DE LOOPS CONTROLADOS (task manager)
-- ═══════════════════════════════════════════════════════════════
local LoopManager = {}
LoopManager.loops = {}

function LoopManager.start(key, fn, interval)
    if LoopManager.loops[key] then
        LoopManager.stop(key)
    end
    interval = interval or 0.1
    local running = true
    LoopManager.loops[key] = {
        stop = function() running = false end,
    }
    task.spawn(function()
        while running and ScreenGui.Parent do
            local ok, err = pcall(fn)
            if not ok then
                warn("[KingOfDark] Loop '" .. key .. "' erro: " .. tostring(err))
            end
            task.wait(interval)
        end
        LoopManager.loops[key] = nil
    end)
end

function LoopManager.stop(key)
    if LoopManager.loops[key] then
        LoopManager.loops[key].stop()
        LoopManager.loops[key] = nil
    end
end

function LoopManager.isRunning(key)
    return LoopManager.loops[key] ~= nil
end

--  [39] UTILITÁRIO: verificar se alvo é válido
-- ═══════════════════════════════════════════════════════════════
local function isValidTarget(target)
    if not target or not target.player or not target.hrp then return false end
    if Combat.killAuraTeamCheck and target.player.Team == LP.Team and LP.Team ~= nil then
        return false
    end
    if target.hum and target.hum.Health <= 0 then return false end
    return true
end

local function hasLineOfSight(target)
    if Combat.killAuraWalls then return true end
    local myHRP = U.getHRP()
    if not myHRP or not target.hrp then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { U.getChar(), target.char }
    local result = workspace:Raycast(myHRP.Position, target.hrp.Position - myHRP.Position, params)
    return result == nil
end

local function doAttack(target)
    if not target then return end
    local char = U.getChar()
    -- Tenta usar tool equipada
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        U.safe(function() tool:Activate() end)
    end
    -- Também tenta via tecla (fallback)
    U.safe(function()
        local vim = game:GetService("VirtualInputManager")
        vim:SendKeyEvent(true, Enum.KeyCode.E, false, game)
        task.wait(0.02)
        vim:SendKeyEvent(false, Enum.KeyCode.E, false, game)
    end)
end

--  [44] FUNÇÃO: ativar skill específica
-- ═══════════════════════════════════════════════════════════════
local function activateSkill(skill)
    if not skill then return end
    local char = U.getChar()
    -- Primeiro tenta via tecla (mais comum em jogos anime)
    U.safe(function()
        local vim = game:GetService("VirtualInputManager")
        local key = Enum.KeyCode[skill.key]
        if key then
            vim:SendKeyEvent(true, key, false, game)
            task.wait(0.03)
            vim:SendKeyEvent(false, key, false, game)
        end
    end)
    -- Fallback: procura tool com nome parecido
    U.safe(function()
        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") and tool.Name:lower():find(skill.name:lower():sub(1, 4)) then
                tool:Activate()
                break
            end
        end
    end)
end

-- ═══════════════════════════════════════════════════════════════
--  [45] FUNÇÃO: equipar personagem (via tools / UI do jogo)
-- ═══════════════════════════════════════════════════════════════
local function equipCharacter(charId)
    -- Fallback universal: procura por botões com o nome do personagem
    U.safe(function()
        local playerGui = LP:FindFirstChildOfClass("PlayerGui")
        if not playerGui then return end
        for _, gui in ipairs(playerGui:GetDescendants()) do
            if gui:IsA("TextButton") or gui:IsA("ImageButton") then
                local name = (gui.Name .. (gui.Text or "")):lower()
                if name:find(charId:lower()) then
                    gui:Activate()
                    return
                end
            end
        end
    end)
end

local function buildCharacterCard(parent, charData)
    local card = Instance.new("TextButton")

    local stroke = Instance.new("UIStroke")

    stroke.Thickness = 1
    stroke.Transparency = 0.7

    tierL.BackgroundTransparency = 1

        U.tw(card, 0.15, { BackgroundColor3 = T.CardHi, BackgroundTransparency = 0.05 })
        U.tw(stroke, 0.15, { Color = charData.color, Transparency = 0.3 })

    end)

--  [50] SISTEMA DE ESP (Drawing API para performance)
-- ═══════════════════════════════════════════════════════════════
local ESP = {}
ESP.objects = {}
ESP.useDrawing = (Drawing ~= nil)

function ESP.createForPlayer(player)
    if ESP.objects[player] then return ESP.objects[player] end

    local obj = {
        player = player,
        box = nil,
        nameLabel = nil,
        distLabel = nil,
        healthBg = nil,
        healthFill = nil,
    }

    if ESP.useDrawing then
        obj.box        = Drawing.new("Square")
        obj.box.Thickness = Visual.espOutlineThickness
        obj.box.Filled = false
        obj.box.Transparency = 1
        obj.box.Color = Visual.espColor

        obj.nameLabel  = Drawing.new("Text")
        obj.nameLabel.Size = 14
        obj.nameLabel.Center = true
        obj.nameLabel.Outline = true
        obj.nameLabel.OutlineColor = Color3.new(0, 0, 0)
        obj.nameLabel.Color = Color3.new(1, 1, 1)
        obj.nameLabel.Transparency = 1

        obj.distLabel  = Drawing.new("Text")
        obj.distLabel.Size = 12
        obj.distLabel.Center = true
        obj.distLabel.Outline = true
        obj.distLabel.OutlineColor = Color3.new(0, 0, 0)
        obj.distLabel.Color = Visual.espColor
        obj.distLabel.Transparency = 1

        obj.healthBg   = Drawing.new("Square")
        obj.healthBg.Filled = true
        obj.healthBg.Color = Color3.new(0, 0, 0)
        obj.healthBg.Transparency = 0.5

        obj.healthFill = Drawing.new("Square")
        obj.healthFill.Filled = true
        obj.healthFill.Color = Color3.fromRGB(70, 220, 130)
        obj.healthFill.Transparency = 1
    end

    ESP.objects[player] = obj
    return obj
end

function ESP.removeForPlayer(player)
    local obj = ESP.objects[player]
    if not obj then return end
    if ESP.useDrawing then
        for _, drawing in pairs(obj) do
            if typeof(drawing) == "Instance" and drawing.Remove then
                pcall(function() drawing:Remove() end)
            end
        end
    end
    ESP.objects[player] = nil
end

function ESP.update()
    for player, obj in pairs(ESP.objects) do
        if not Visual.espEnabled then
            if ESP.useDrawing then
                for _, d in pairs(obj) do
                    if typeof(d) == "Instance" and d.Visible ~= nil then
                        pcall(function() d.Visible = false end)
                    end
                end
            end
            continue
        end

        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then
            if ESP.useDrawing then
                for _, d in pairs(obj) do
                    if typeof(d) == "Instance" and d.Visible ~= nil then
                        pcall(function() d.Visible = false end)
                    end
                end
            end
            continue
        end

        local myHRP = U.getHRP()
        if not myHRP then continue end
        local dist = (hrp.Position - myHRP.Position).Magnitude
        if dist > Visual.espMaxDistance then
            if ESP.useDrawing then
                for _, d in pairs(obj) do
                    if typeof(d) == "Instance" and d.Visible ~= nil then
                        pcall(function() d.Visible = false end)
                    end
                end
            end
            continue
        end

        if Visual.espTeamCheck and player.Team == LP.Team and LP.Team ~= nil then
            if ESP.useDrawing then
                for _, d in pairs(obj) do
                    if typeof(d) == "Instance" and d.Visible ~= nil then
                        pcall(function() d.Visible = false end)
                    end
                end
            end
            continue
        end

        -- Posição na tela
        local hrpPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
        local headPos = Camera:WorldToViewportPoint((hrp.Position + Vector3.new(0, 3, 0)))

        if not onScreen and hrpPos.Z < 0 then
            if ESP.useDrawing then
                for _, d in pairs(obj) do
                    if typeof(d) == "Instance" and d.Visible ~= nil then
                        pcall(function() d.Visible = false end)
                    end
                end
            end
            continue
        end

        local height = math.abs(headPos.Y - hrpPos.Y) * 2
        local width = height * 0.6

        if ESP.useDrawing then
            local topLeft = Vector2.new(hrpPos.X - width / 2, hrpPos.Y - height / 2)
            local bottomRight = Vector2.new(hrpPos.X + width / 2, hrpPos.Y + height / 2)

            -- Box
            if obj.box then
                obj.box.Visible = Visual.espBox
                obj.box.Size = Vector2.new(width, height)
                obj.box.Position = topLeft
                obj.box.Color = Visual.espColor
                obj.box.Thickness = Visual.espOutlineThickness
            end

            -- Nome
            if obj.nameLabel then
                obj.nameLabel.Visible = Visual.espName
                obj.nameLabel.Text = player.Name
                obj.nameLabel.Position = Vector2.new(hrpPos.X, topLeft.Y - 18)
                obj.nameLabel.Color = Color3.new(1, 1, 1)
            end

            -- Distância
            if obj.distLabel then
                obj.distLabel.Visible = Visual.espDistance
                obj.distLabel.Text = string.format("[%d m]", math.floor(dist))
                obj.distLabel.Position = Vector2.new(hrpPos.X, bottomRight.Y + 2)
                obj.distLabel.Color = Visual.espColor
            end

            -- Health bar
            if obj.healthBg and obj.healthFill then
                obj.healthBg.Visible = Visual.espHealthBar
                obj.healthFill.Visible = Visual.espHealthBar
                local barX = topLeft.X - 8
                local barW = 4
                local barH = height
                obj.healthBg.Size = Vector2.new(barW, barH)
                obj.healthBg.Position = Vector2.new(barX, topLeft.Y)
                local hpPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                local fillH = barH * hpPct
                obj.healthFill.Size = Vector2.new(barW, fillH)
                obj.healthFill.Position = Vector2.new(barX, topLeft.Y + (barH - fillH))
                obj.healthFill.Color = hpPct > 0.5 and Color3.fromRGB(70, 220, 130)
                    or (hpPct > 0.25 and Color3.fromRGB(255, 185, 65) or Color3.fromRGB(255, 75, 95))
            end
        else
            -- Fallback: Highlight Instance (sem Drawing API)
            if not obj.highlight then
                local hl = Instance.new("Highlight")
                hl.Name = "KoD_ESP"
                hl.FillColor = Visual.espColor
                hl.FillTransparency = Visual.espFillTransparency
                hl.OutlineColor = Visual.espColor
                hl.OutlineTransparency = 0
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Adornee = char
                hl.Parent = ScreenGui
                obj.highlight = hl
            end
            if obj.highlight then
                obj.highlight.Adornee = char
                obj.highlight.FillColor = Visual.espColor
                obj.highlight.OutlineColor = Visual.espColor
            end
        end
    end
end

local ChamsHighlights = {}

local function updateChams()

        if not char then
            if ChamsHighlights[p] then
                ChamsHighlights[p]:Destroy()

            continue

        if not Visual.chamsEnabled then
            if ChamsHighlights[p] then
                ChamsHighlights[p]:Destroy()
                ChamsHighlights[p] = nil
            end
            continue
        end
        if not ChamsHighlights[p] then
            local hl = Instance.new("Highlight")
            hl.Name = "KoD_Chams"
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Adornee = char
            hl.Parent = ScreenGui
            ChamsHighlights[p] = hl
        end
        local hl = ChamsHighlights[p]
        hl.FillColor = Visual.chamsColor
        hl.FillTransparency = Visual.chamsTransparency
        hl.OutlineColor = Visual.chamsColor
        hl.OutlineTransparency = 0.3
    end
end

local LightingDefaults = {
    Brightness = Lighting.Brightness,
    ClockTime  = Lighting.ClockTime,
    FogEnd     = Lighting.FogEnd,
    FogStart   = Lighting.FogStart,
    Ambient    = Lighting.Ambient,
    OutdoorAmbient = Lighting.OutdoorAmbient,
    GlobalShadows = Lighting.GlobalShadows,
}

local function applyFullbright(state)
    if state then
        Lighting.Brightness = 3
        Lighting.ClockTime = 14
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 1e6
        Lighting.FogStart = 0
    else
        Lighting.Brightness = LightingDefaults.Brightness
        Lighting.ClockTime = LightingDefaults.ClockTime
        Lighting.GlobalShadows = LightingDefaults.GlobalShadows
        Lighting.FogEnd = LightingDefaults.FogEnd
        Lighting.FogStart = LightingDefaults.FogStart
    end
end

local function applyNoFog(state)
    if state then
        Lighting.FogEnd = 1e6
    else
        Lighting.FogEnd = LightingDefaults.FogEnd
    end
end

local XRayCache = {}

local function updateXRay()
    if Visual.xrayEnabled then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Material ~= Enum.Material.ForceField then
                if not XRayCache[obj] then
                    XRayCache[obj] = {
                        Transparency = obj.Transparency,
                        Material = obj.Material,
                    }
                end
                obj.Transparency = 0.85
            end
        end
    else
        for obj, data in pairs(XRayCache) do
            if obj.Parent then
                pcall(function()
                    obj.Transparency = data.Transparency
                    obj.Material = data.Material
                end)
            end
        end
        XRayCache = {}
    end
end

--  [62] FLY (WASD + Mouse + Vertical)
-- ═══════════════════════════════════════════════════════════════
local flyConnections = {}
local flyVelocity = Vector3.zero
local flyAngular = Vector3.zero

local function stopFly()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    if flyBodyVel then flyBodyVel:Destroy(); flyBodyVel = nil end
    if flyBodyGyro then flyBodyGyro:Destroy(); flyBodyGyro = nil end

    for _, c in ipairs(flyConnections) do
        if c.Disconnect then c:Disconnect() end
    end
    flyConnections = {}

    U.safe(function()
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end)
end

local function startFly()
    stopFly()
    local char = U.getChar()
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if not hrp then return end
    local hum = char:FindFirstChildOfClass("Humanoid")

    local bv = Instance.new("BodyVelocity")
    bv.Name = "KoD_FlyVel"
    bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    bv.Velocity = Vector3.zero
    bv.P = 1250
    bv.Parent = hrp
    flyBodyVel = bv

    local bg = Instance.new("BodyGyro")
    bg.Name = "KoD_FlyGyro"
    bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
    bg.P = 3000
    bg.D = 50
    bg.CFrame = hrp.CFrame
    bg.Parent = hrp
    flyBodyGyro = bg

    if hum then
        hum.PlatformStand = true
    end

    -- Conexão de movimento (RenderStepped)
    table.insert(flyConnections, RunService.RenderStepped:Connect(function(dt)
        if not Movement.flyEnabled then return end
        local camera = workspace.CurrentCamera
        local moveDir = Vector3.zero

        if Movement.flyMode == "WASD" then
            local forward = camera.CFrame.LookVector
            local right = camera.CFrame.RightVector

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                moveDir = moveDir + forward
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                moveDir = moveDir - forward
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                moveDir = moveDir + right
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                moveDir = moveDir - right
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                moveDir = moveDir + Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveDir = moveDir - Vector3.new(0, 1, 0)
            end
        elseif Movement.flyMode == "Segir Mouse" then
            moveDir = camera.CFrame.LookVector
        elseif Movement.flyMode == "Vertical" then
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                moveDir = Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveDir = Vector3.new(0, -1, 0)
            end
        end

        if moveDir.Magnitude > 0 then
            moveDir = moveDir.Unit * Movement.flySpeed
        end
        bv.Velocity = moveDir
        bg.CFrame = camera.CFrame
    end))
end

local function applyNoclip()
    local char = LP.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide then
            if Movement.noclipEnabled then
                if not part:GetAttribute("KoD_OrigCollide") then
                    part:SetAttribute("KoD_OrigCollide", true)
                end
                part.CanCollide = false
            else
                if part:GetAttribute("KoD_OrigCollide") then
                    part.CanCollide = true
                    part:SetAttribute("KoD_OrigCollide", nil)
                end
            end
        end
    end
end

LoopManager.start("noclipLoop", function()
    if Movement.noclipEnabled then applyNoclip() end
end, 0.2)

local lastDash = 0
local function performDash()
    if tick() - lastDash < Movement.dashCooldown then return end
    local hrp = U.getHRP()
    if not hrp then return end
    lastDash = tick()
    local direction = Vector3.new(hrp.Velocity.X, 0, hrp.Velocity.Z)
    if direction.Magnitude < 1 then
        direction = Camera.CFrame.LookVector * Vector3.new(1, 0, 1)
    end
    direction = direction.Unit
    hrp.AssemblyLinearVelocity = direction * Movement.dashSpeed + Vector3.new(0, 10, 0)
end

LoopManager.start("walkWaterLoop", function()
    if not Movement.walkWater then return end

--  [73] FPS BOOST (limpa decorações)
-- ═══════════════════════════════════════════════════════════════
local FpsBoostCache = {}
local function applyFpsBoost(level)
    level = level or 1

    -- Restaurar tudo
    for obj, data in pairs(FpsBoostCache) do
        if obj.Parent then
            pcall(function()
                if data.CastShadow ~= nil then obj.CastShadow = data.CastShadow end
                if data.Material ~= nil and obj.Material ~= nil then obj.Material = data.Material end
            end)
        end
    end
    FpsBoostCache = {}

    if not Utility.fpsBoostEnabled then return end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Decal") or obj:IsA("Texture") then
            if not obj:GetAttribute("KoD_FpsCached") then
                FpsBoostCache[obj] = {
                    CastShadow = obj.CastShadow,
                    Material = obj.Material,
                }
                obj:SetAttribute("KoD_FpsCached", true)
            end
            pcall(function()
                if level >= 1 then
                    if obj:IsA("BasePart") then obj.CastShadow = false end
                end
                if level >= 2 then
                    if obj:IsA("Decal") or obj:IsA("Texture") then obj.Transparency = 1 end
                end
                if level >= 3 then
                    if obj:IsA("BasePart") and obj.Material ~= Enum.Material.SmoothPlastic then
                        obj.Material = Enum.Material.SmoothPlastic
                    end
                end
            end)
        end
    end
end

-- Remover efeitos

                obj.Enabled = false
            end
        end
    end
end, 3)

function ConfigManager.apply(data)
    if not data or type(data) ~= "table" then return end

    -- Tema
    if data.theme and ThemePresets[data.theme] then
        applyTheme(data.theme)
        U.safe(function()
            MainStroke.Color = T.Accent
            LogoText.TextColor3 = T.Accent
            BrandTitle.TextColor3 = T.Accent
            RedLine.BackgroundColor3 = T.Accent
        end)
    end

    -- UI
    if data.ui then
        if data.ui.scale then
            KOD.uiScale = data.ui.scale
            Main.Size = UDim2.new(0, 860 * data.ui.scale, 0, 600 * data.ui.scale)
        end
        if data.ui.transparency then
            KOD.uiTransparency = data.ui.transparency
            Main.BackgroundTransparency = data.ui.transparency
        end
        if data.ui.floatBtnVisible ~= nil then
            FloatBtn.Visible = data.ui.floatBtnVisible
        end
    end

    -- Combate
    if data.combat then
        for k, v in pairs(data.combat) do
            if Combat[k] ~= nil then Combat[k] = v end
        end
    end

    -- Visual
    if data.visual then
        for k, v in pairs(data.visual) do
            if Visual[k] ~= nil then Visual[k] = v end
        end
        U.safe(function()
            applyFullbright(Visual.fullbright)
            applyNoFog(Visual.noFog)
            applyFOV()
            applyZoom()
            updateCrosshair()
            updateWatermark()
            if Visual.skyboxEnabled then applySkybox(Visual.skyboxPreset) end
        end)
    end

    -- Movimento
    if data.movement then
        for k, v in pairs(data.movement) do
            if Movement[k] ~= nil then Movement[k] = v end
        end
        U.safe(function()
            applySpeed()
            applyJump()
            if Movement.flyEnabled then startFly() end
            if Movement.noclipEnabled then applyNoclip() end
        end)
    end

    -- Utilidades
    if data.utility then
        for k, v in pairs(data.utility) do
            if Utility[k] ~= nil then Utility[k] = v end
        end
    end
end

function ConfigManager.save()
    local data = ConfigManager.collect()
    Save.data = data
    Save.write()
    return data
end

function ConfigManager.load()
    local data = Save.load()
    if data and next(data) ~= nil then
        ConfigManager.apply(data)
        return data
    end
    return nil
end

function ConfigManager.reset()
    Save.data = {}
    Save.write()
end

function ConfigManager.export()
    local data = ConfigManager.collect()
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    if not ok then return nil end
    -- Base64 simples
    local ok2, b64 = pcall(function()
        return HttpService:Base64Encode(encoded)
    end)
    if ok2 then return b64 end
    return encoded
end

function ConfigManager.import(str)
    if not str or str == "" then return false end
    local ok, decoded = pcall(function()
        return HttpService:Base64Decode(str)
    end)
    if ok then
        local ok2, data = pcall(HttpService.JSONDecode, HttpService, decoded)
        if ok2 and data then
            ConfigManager.apply(data)
            return true
        end
    end
    -- Fallback: tentar direto JSON
    local ok3, data2 = pcall(HttpService.JSONDecode, HttpService, str)
    if ok3 and data2 then
        ConfigManager.apply(data2)
        return true
    end
    return false
end

    end,

local function hash(str)
    local h = 0
    for i = 1, #str do
        h = (h * 31 + string.byte(str, i)) % 2147483647
    end
    return tostring(h)
end

local oldNamecall

if hookmetamethod then
    local mt = getrawmetatable(game)
    if mt then
        local oldIndex = mt.__index
        oldNamecall = mt.__namecall
        setreadonly(mt, false)

        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = { ... }

            -- Intercepta chamadas de dano
            if Premium.ohk and method == "FireServer" and typeof(self) == "Instance" then
                local selfName = self.Name:lower()
                if selfName:find("damage")
                or selfName:find("hit")
                or selfName:find("attack")
                or selfName:find("combat") then
                    -- Multiplica valores numéricos por 999
                    for i, v in ipairs(args) do
                        if type(v) == "number" and v > 0 and v < 10000 then
                            args[i] = v * 999
                        end
                    end
                    return oldNamecall(self, table.unpack(args))
                end
            end

            -- Damage Multiplier (customizado)
            if Premium.damageMultiplier > 1 and method == "FireServer" and typeof(self) == "Instance" then
                local selfName = self.Name:lower()
                if selfName:find("damage") or selfName:find("hit") then
                    for i, v in ipairs(args) do
                        if type(v) == "number" and v > 0 and v < 10000 then
                            args[i] = v * Premium.damageMultiplier
                        end
                    end
                    return oldNamecall(self, table.unpack(args))
                end
            end

            -- Anti-Kick: intercepta Kick remotas
            if Premium.antiKick and method == "FireServer" and typeof(self) == "Instance" then
                local selfName = self.Name:lower()
                if selfName:find("kick") or selfName:find("ban") then
                    return nil
                end
            end

            return oldNamecall(self, ...)
        end)

        setreadonly(mt, true)
        U.log("Premium", "Namecall hook instalado com sucesso.")
    end
end

local function applyInvisible(state)
    local char = LP.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if state then
                if not part:GetAttribute("KoD_OrigTrans") then
                    part:SetAttribute("KoD_OrigTrans", part.Transparency)
                end
                part.Transparency = 1
            else
                local orig = part:GetAttribute("KoD_OrigTrans")
                if orig then
                    part.Transparency = orig
                    part:SetAttribute("KoD_OrigTrans", nil)
                end
            end
        end
        if part:IsA("Decal") or part:IsA("Texture") then
            if state then
                if not part:GetAttribute("KoD_OrigTrans") then
                    part:SetAttribute("KoD_OrigTrans", part.Transparency)
                end
                part.Transparency = 1
            else
                local orig = part:GetAttribute("KoD_OrigTrans")
                if orig then
                    part.Transparency = orig
                    part:SetAttribute("KoD_OrigTrans", nil)
                end
            end
        end
    end
end

local Plugins = {
    loaded = {},
    folder = "KingOfDark/plugins/",
}

function Plugins.list()
    if not listfiles then return {} end
    local ok, files = pcall(listfiles, Plugins.folder)
    if not ok then return {} end
    local lua = {}
    for _, f in ipairs(files) do
        if f:sub(-4) == ".lua" or f:sub(-6) == ".lua.txt" then
            table.insert(lua, f)
        end
    end
    return lua
end

function Plugins.load(path)
    if not readfile then return false, "Executor sem readfile" end
    local ok, content = pcall(readfile, path)
    if not ok or not content then return false, "Falha ao ler arquivo" end
    local fn, err = loadstring(content)
    if not fn then return false, "Erro de sintaxe: " .. tostring(err) end
    local success, runErr = pcall(fn)
    if not success then return false, "Erro ao executar: " .. tostring(runErr) end
    Plugins.loaded[path] = true
    return true, "Plugin carregado"
end

local Log = {
    entries = {},
    max = 500,
    listeners = {},
    levels = {
        info    = { color = T.Accent,   tag = "INFO",  icon = "◆" },
        success = { color = T.Success,  tag = "OK",    icon = "✔" },
        warn    = { color = T.Warning,  tag = "WARN",  icon = "⚠" },
        error   = { color = T.Danger,   tag = "ERR",   icon = "✕" },
        debug   = { color = T.TextMuted,tag = "DBG",   icon = "◦" },
    },
}

function Log.push(level, msg)
    local lvl = Log.levels[level] or Log.levels.info
    local entry = {
        level = level,
        msg = tostring(msg),
        time = os.date("%H:%M:%S"),
        color = lvl.color,
        tag = lvl.tag,
        icon = lvl.icon,
    }
    table.insert(Log.entries, entry)
    if #Log.entries > Log.max then
        table.remove(Log.entries, 1)
    end
    -- Notificar listeners
    for _, cb in ipairs(Log.listeners) do
        U.safe(cb, entry)
    end
    print(string.format("[KingOfDark] [%s] %s", entry.tag, entry.msg))
end

function Log.on(cb)
    table.insert(Log.listeners, cb)
end

function Log.getRecent(n)
    n = n or 100
    local out = {}
    local start = math.max(1, #Log.entries - n + 1)
    for i = start, #Log.entries do
        table.insert(out, Log.entries[i])
    end
    return out
end

function Log.clear()
    Log.entries = {}
end

-- Redirecionar U.log para o novo sistema
local function _ULog(tag, msg)
    Log.push("info", string.format("[%s] %s", tag, msg))
end

    end

    -- Destruir efeitos
    if PremiumUI.blur then
        PremiumUI.blur:Destroy()
    end

    -- Restaurar iluminação
    pcall(function()
        applyFullbright(false)
        applyNoFog(false)
        applySkybox("Nenhum")
        applyFOV()
        applyZoom()
    end)

    -- Restaurar personagem
    pcall(function()
        stopFly()
        Movement.noclipEnabled = false
        applyNoclip()
        applySpeed()
        applyJump()
        applyHipHeight()
    end)

    -- Remover ESP
    if ESP.useDrawing then
        for _, obj in pairs(ESP.objects) do
            for _, d in pairs(obj) do
                if typeof(d) == "Instance" and d.Remove then
                    pcall(function() d:Remove() end)
                end
            end
        end
    end

    Log.push("success", "Cleanup finalizado. Até a próxima!")
end
